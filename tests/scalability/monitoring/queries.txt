# --- policy agent (kube-network-policies) ------------------------------------
agent_cpu_cores|sum by (node) (rate(container_cpu_usage_seconds_total{namespace="kube-system",pod=~"kube-network-policies.*",container!=""}[30s]))
agent_memory_wss_bytes|sum by (node) (container_memory_working_set_bytes{namespace="kube-system",pod=~"kube-network-policies.*",container!=""})
agent_memory_rss_bytes|sum by (node) (container_memory_rss{namespace="kube-system",pod=~"kube-network-policies.*",container!=""})
agent_verdicts_per_s|sum by (node,verdict) (rate(kube_network_policies_packet_count[30s]))
agent_process_p50_us|max by (node) (kube_network_policies_packet_process_duration_microseconds{quantile="0.5"})
agent_process_p99_us|max by (node) (kube_network_policies_packet_process_duration_microseconds{quantile="0.99"})
agent_process_hist_p99_us|histogram_quantile(0.99, sum by (node,le) (rate(kube_network_policies_packet_process_time_bucket[30s])))
agent_queue_depth|max by (node,queue) (kube_network_policies_nfqueue_queue_total)
agent_queue_dropped|max by (node,queue) (kube_network_policies_nfqueue_queue_dropped)
agent_user_dropped|max by (node,queue) (kube_network_policies_nfqueue_user_dropped)
agent_go_heap_bytes|sum by (node) (go_memstats_heap_inuse_bytes{job="kube-network-policies"})
agent_go_goroutines|sum by (node) (go_goroutines{job="kube-network-policies"})

# --- metadata distributor (kube-ip-tracker), present only for the iptracker DUT ---

iptracker_memory_wss_bytes|sum(container_memory_working_set_bytes{namespace="kube-system",pod=~"kube-ip-tracker.*",container!=""})
